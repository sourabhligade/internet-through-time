### Gate helper (extras)

```js
function hasConsent() {
  var raw = loadJSON(key("gdpr"), null);
  return !!(raw && raw.real && raw.multiStep);
}
function paintBanner(doc) {
  doc.querySelectorAll("[data-itt18-needs-consent]").forEach(function (root) {
    if (hasConsent()) {
      var b = root.querySelector("[data-itt18-cmp]");
      if (b) b.setAttribute("hidden", "");
      root.querySelectorAll("[data-itt18-primary]").forEach(function (el) {
        el.removeAttribute("disabled");
      });
    } else {
      root.querySelectorAll("[data-itt18-primary]").forEach(function (el) {
        el.setAttribute("disabled", "disabled");
      });
    }
  });
}
```
